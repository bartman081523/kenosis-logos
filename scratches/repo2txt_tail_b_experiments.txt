"""
smoke_hermeneutik.py — SP8: der hermeneutische Zirkel Python <-> Lisp.

Die Review-Frage: compile_rule (composite.py:76-88) erkennt nur den EXAKTEN
Tautologie-Knoten ('identity','x') — aber die algebraisch trivialen Identitaeten
add(0,x) / multiply(1,x) / add(x,0) / multiply(x,1) haben KEINEN identity-Knoten
und laufen als harmlose HALT-Kette durch. Die Lisp-Reflexion (Vendor-Reduktor
symbolic-operator.lisp:118-182 ueber die Bruecke, Op simplify_ast) wascht den
identity-Wrapper weg und normalisiert die Form; die Entlarvungs-Schicht des
Bridge-Ops markiert, wenn die Reduktion auf das nackte Var-Blatt x fuhrt UND
eine triviale Identitaet im Ursprung stand.

Phase 1 — Paritaet: dieselben Faelle durch Python-Seite
          (lisp_parity.simplify_ast_canonical, Vendor-Spiegel) UND Lisp-Seite
          (SBCL handle-simplify-ast); exakte Feldgleichheit Pflicht.
Phase 2 — der Zirkel: die entlarvte Tautologie wird zum Programm —
          compile_rule uebersetzt ('identity','x') zur Selbst-Transition,
          Turing erkennt den Loop, die Komposition expandiert die Basis
          (uni_256:113, TAUT_POOL6). Kontrast: kein einfacher sin(x)-Ast
          (+0) compiliert zum Loop — HALT-Garantie.

Befunde:
  sp8_parity   — Python- und Lisp-Seite liefern identische Felder je Fall
  sp8_reveal   — die Entlarvung trifft exakt die Tautologie-Familie:
                 add(0,x)/multiply(1,x)/add(x,0)/multiply(x,1)/identity(x)
                 -> revealed; sin(x)+0 / x+x / const-Fold / 0*x -> nicht
  sp8_zirkel   — entlarvte Tautologie erzwungen (Schritt 1, TAUT_POOL6):
                 LOOP-Verdikt, Expansion "step1:LOOP", Basis > 3, Gaps endlich
  sp8_contrast — sin(x)+0 (nicht entlarvt) erzwungen: kein LOOP in 6 Schritten,
                 Basis bleibt 3 (HALT-Garantie des Nicht-Tautologen)
  sp8_finite   — alle Gaps endlich
  sp7_break    — Kripke Plus/Quus: die erzwungene Tautologie an Schritt 9 wird
                 Programm (LOOP-Verdikt an 9, Expansion "step9:LOOP", Basis > 3)
  sp7_paradigm — das Gödel-Paradigmen-Signal (ratio > 2) feuert im Fenster der
                 Schritte 9-13 — erst NACH wiederholtem Gebrauch der Regel
  sp7_quus     — dieselbe Stelle, harmlose Form sin(x)+0: kein LOOP an 9,
                 keine "step9"-Expansion — gleicher Gebrauch, andere Bedeutung

Run:  python3 -m scratches.smoke_hermeneutik          # CPU, braucht SBCL
Log:  ... | tee scratches/hermeneutik_experiment.log

TAUTAST = ("identity", "x")

                             "EPSILON", "ZETA"]}

# (name, ast, revealed_expected) — die Paritaets- und Entlarvungs-Matrix.
PARITY_CASES: List[Tuple[str, List[Any], bool]] = [
    ("add0x",      ["add", 0.0, "x"], True),
    ("mul1x",      ["multiply", 1.0, "x"], True),
    ("id_add0x",   ["add", ["identity", "x"], 0.0], True),
    ("sini_add0",  ["add", ["sin", "x"], 0.0], False),
    ("add_xx",     ["add", "x", "x"], False),

    ("mulx1",      ["multiply", "x", 1.0], True),
    ("sub_nofold", ["subtract", 2.0, 1.0], False),
    ("add_fold",   ["add", 2.0, 3.0], False),
    ("mul0annih",  ["multiply", 0.0, ["sin", "x"]], False),
    ("nested_mix", ["add", ["add", ["sin", "x"], 0.0],
                    ["multiply", "x", 1.0]], False),
    ("pure_id",    ["identity", "x"], True),
]


def compare_case(name: str, ast: List[Any],
                 python: Dict[str, Any], lisp: Dict[str, Any],
                 revealed_expected: Any) -> Dict[str, Any]:
    fields = ("ast_simplified", "size_in", "size_out")
    equal = all(_same(python[f], lisp[f]) for f in fields)
    equal &= python.get("revealed") == lisp.get("revealed")
    equal &= (python.get("contains_trivial_identity")
              == lisp.get("contains_trivial_identity"))
    return {"name": name, "ast": ast, "python": python, "lisp": lisp,
            "equal": equal,
            "revealed": bool(python.get("revealed")),
            "revealed_expected": revealed_expected,
            "flagged": bool(python.get("contains_trivial_identity"))}


def _same(a: Any, b: Any) -> bool:
    if isinstance(a, (int, float)) and isinstance(b, (int, float)):
        return float(a) == float(b)
    if isinstance(a, str) != isinstance(b, str):
        return False
    if isinstance(a, (list, tuple)) and isinstance(b, (list, tuple)):
        return len(a) == len(b) and all(_same(x, y) for x, y in zip(a, b))
    return a == b


# -- Phase 2: der Zirkel ------------------------------------------------------
def force_rule_at(machine: CompositeMachine, ast: Tuple[Any, ...],
                  steps: set) -> None:
    """Probe-Pfad wie smoke_composite_oracle.py — erzwinge `ast` als erste
    Regel des Batch an den genannten Schritten (1-basiert)."""
    wm = machine.wittgenstein
    original = wm.create_rule
    used: set = set()

    def patched() -> GeneratedSymbolicRule:
        current = wm._step_index + 1
        if current in steps and current not in used:
            used.add(current)
            rule = GeneratedSymbolicRule(ast)
            wm.generated_rules.append(rule)
            return rule
        return original()

    wm.create_rule = patched


def run_composite(force_ast: Tuple[Any, ...], steps_run: int,
                  force_steps: set | None = None, seed: int = 42) -> Dict[str, Any]:
    """SP8/SP7-Laufgeruest. Seed dokumentiert: 42 für den SP8-Zirkel
    (natürliche Tautologien bei 10/12), 7 für SP7 (Paradigmen-Signal im
    Fenster 9-13, Seed-Scan 2026-10-25)."""
    machine = CompositeMachine(MachineConfig(seed=seed, mode="in_loop",
                                             params=dict(TAUT_POOL6)))
    force_rule_at(machine, force_ast, force_steps or {1})
    verdicts: List[str] = []
    basis: List[int] = []
    gaps: List[float] = []
    paradigms: List[Dict[str, Any]] = []
    paradigms_c: List[Dict[str, Any]] = []
    for _ in range(steps_run):
        result = machine.step(None)
        verdicts.append(str(result.invariants["verdict"]))
        basis.append(len(machine.turing.basis))
        gaps.append(float(result.gap))
        paradigms.append(dict(result.invariants.get("paradigm_signal", {})))
        paradigms_c.append(dict(result.invariants.get("paradigm_signal_composite", {})))
    return {"verdicts": verdicts, "basis": basis, "final_basis": basis[-1],
            "expansions": list(machine.expansions), "gaps": gaps,
            "gaps_finite": all(g == g and abs(g) != float("inf") for g in gaps),
            "paradigm_ratio": [p.get("ratio") for p in paradigms],
            "paradigm_detected": [bool(p.get("detected")) for p in paradigms_c],
            "forced_ast": list(force_ast)}


def main() -> int:
    bridge = LISPBridge()
    if not bridge.available():
        print("SKIP — SBCL-Bridge nicht verfuegbar (Exit 3, kein stiller False-Pass)")
        return 3

    print("[SP8.1] Paritaet Python (lisp_parity) vs Lisp (SBCL simplify_ast) …")
    rows: List[Dict[str, Any]] = []
    for name, ast, revealed_expected in PARITY_CASES:
        python = simplify_ast_canonical(ast)
        lisp = bridge.simplify_ast(ast)
        row = compare_case(name, ast, python, lisp, revealed_expected)
        rows.append(row)
    # Maschinen-Ursprung: ein echter Sampler-AST wandert durch den Zirkel.
    def ast_to_payload(ast: Any) -> Any:
        """Python-Tupel-AST -> JSON-Payload (Rekursion; bares Blatt bleibt)."""
        if isinstance(ast, tuple):
            return [ast_to_payload(item) for item in ast]
        return ast

    machine_probe = CompositeMachine(MachineConfig(seed=11, mode="in_loop"))
    sampled = machine_probe.wittgenstein.create_rule()
    ast_machine = ast_to_payload(sampled.ast)
    python_m = simplify_ast_canonical(ast_machine)
    lisp_m = bridge.simplify_ast(ast_machine)
    rows.append(compare_case("machine_seed11", ast_machine, python_m, lisp_m,
                             revealed_expected=None))
    # Sampler-Fall: nur Paritaetsanspruch (kein Reveal-Erwartungswert).

    print("[SP8.2] Der Zirkel: entlarvte Tautologie -> Programm -> LOOP …")
    zirkel = run_composite(TAUTAST, STEPS)

    print("[SP8.3] Kontrast: sin(x)+0 (nicht entlarvt) -> HALT-Garantie …")
    contrast = run_composite(("add", ("sin", "x"), 0.0), STEPS_CONTRAST)

    print("[SP7] Kripke Plus/Quus: erzwungene Tautologie an Schritt 9 (seed 7) …")
    kripke = run_composite(TAUTAST, STEPS, force_steps={9}, seed=7)
    print("[SP7.2] Kontrast (Quus): sin(x)+0 an Schritt 9 — keine Bruchstelle …")
    contrast_q = run_composite(("add", ("sin", "x"), 0.0), STEPS,
                               force_steps={9}, seed=7)
    first_detect = next((idx for idx, flag in
                         enumerate(kripke["paradigm_detected"], start=1) if flag),
                        None)

    print("=" * 88)
    print("SP8 — HERMENEUTISCHER ZIRKEL (PYTHON <-> LISP)")
    print("=" * 88)
    for row in rows:
        expected = ("—" if row.get("revealed_expected") is None
                    else str(row["revealed_expected"]))
        print(f"  {row['name']:14s} simp={json.dumps(row['python']['ast_simplified']):22s}"
              f" parity={row['equal']!s:5s} reveal={row['revealed']!s:5s} "
              f"(erwartet {expected:5s}) "
              f"sizes=({row['python']['size_in']},{row['python']['size_out']})")
    print(f"  Zirkel: verdicts={zirkel['verdicts'][:6]} … basis_end={zirkel['final_basis']} "
          f"expansions={zirkel['expansions']}")
    print(f"  Kontrast: verdicts={contrast['verdicts']} basis={contrast['basis']}")
    print(f"  Kripke: verdicts(9..13)={kripke['verdicts'][8:13]} "
          f"expansions={kripke['expansions']} basis_end={kripke['final_basis']}")
    print(f"  paradigm_detected={kripke['paradigm_detected']} "
          f"first_detect={first_detect}")
    print(f"  Quus-Kontrast: verdicts(9..13)={contrast_q['verdicts'][8:13]} "
          f"expansions={contrast_q['expansions']} "
          f"basis_end={contrast_q['final_basis']}")

    findings: List[Tuple[str, bool]] = [
        ("sp8_parity", all(r["equal"] for r in rows)),
        ("sp8_reveal", all(r["revealed"] == r["revealed_expected"] for r in rows
                           if r["revealed_expected"] is not None)),
        ("sp8_zirkel", "LOOP" in zirkel["verdicts"]
                       and "step1:LOOP" in zirkel["expansions"]
                       and zirkel["final_basis"] > 3 and zirkel["gaps_finite"]),
        ("sp8_contrast", "LOOP" not in contrast["verdicts"]
                         and contrast["final_basis"] == 3),
        ("sp8_finite", zirkel["gaps_finite"] and contrast["gaps_finite"]),
        ("sp7_break", kripke["verdicts"][8] == "LOOP"
                      and "step9:LOOP" in kripke["expansions"]
                      and kripke["final_basis"] > 3 and kripke["gaps_finite"]),
        ("sp7_paradigm", first_detect is not None and 9 <= first_detect <= 13),
        ("sp7_quus", contrast_q["verdicts"][8] != "LOOP"
                     and "step9:LOOP" not in contrast_q["expansions"]
                     and contrast_q["gaps_finite"]),
    ]
    print("-" * 88)
    print("BEFUNDE")
    for name, ok in findings:
        print(f"  {name:14s} {'PASS' if ok else 'FAIL'}")

    out = REPO_ROOT / "scratches" / "hermeneutik_experiment.json"
    out.write_text(json.dumps({
        "parity_rows": rows, "zirkel": zirkel, "contrast": contrast,
        "kripke": kripke, "contrast_quus": contrast_q,
        "first_detect": first_detect,
        "findings": [{"name": n, "status": "PASS" if ok else "FAIL"}
                     for n, ok in findings],
    }, indent=2, ensure_ascii=False, default=str), encoding="utf-8")
    print(f"[Save] {out.relative_to(REPO_ROOT)}")
    failed = [n for n, ok in findings if not ok]
    if failed:
        print(f"FEHLGESCHLAGEN: {len(failed)} — {', '.join(failed)}")
        return 1
    print("GRUEN — alle SP8-Befunde bestanden.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def mark(name: str, cond: bool) -> None:
    findings.append((name, "PASS" if cond else "FAIL"))


def raw_res(tokenizer: GeometricTokenizer, w1: str, w2: str) -> float:
    """Paar-Resonanz UNGERUNDET (für Bands auf 1e-12-Niveau)."""
    return float(tokenizer.resonance(tokenizer.embed_word(w1),
                                     tokenizer.embed_word(w2)))


def round6(values: Any) -> Any:
    if isinstance(values, (list, tuple)):
        return [round6(v) for v in values]
    return round(float(values), 6)


def max_dip_after_peak(traj: List[float]) -> float:
    """Größter Rückfall unter dem laufenden Maximum (Monotonie-Maß)."""
    running_max = -1e9
    worst = 0.0
    for r in traj:
        if r > running_max:
            running_max = r
        worst = max(worst, running_max - r)
    return worst


def first_at_threshold(traj: List[float], threshold: float = THRESH
                       ) -> Optional[int]:
    """1-basierte Episode des ersten c >= threshold (None: nie)."""
    for i, r in enumerate(traj):
        if r >= threshold:
            return i + 1
    return None


def bisector_cos(e_a: List[float], e_b: List[float],
                 observer: List[float]) -> float:
    """cos(observer, n(e_a + e_b)) — der Halbierenden-Vorhersagewert des
    symmetrischen Mixing-Fixpunkts (ex ante aus den Gematria-Embeddings)."""
    m = GeometricTokenizer._normalize([x + y for x, y in zip(e_a, e_b)])
    return float(np.dot(observer, m))


def lane_regression() -> Dict[str, Any]:
    """SP9.0 — SP6-Lane exakt nachgestellt; Bit-Delta gegen die Artefakt-
    ziffern. Kein Maschinen-Drift durch die SP9-Runde."""
    machine = WittgensteinMachine(MachineConfig(seed=9, mode="in_loop"))
    tokenizer = machine.tokenizer

    def res(pair: Tuple[str, str]) -> float:
        return round6(raw_res(tokenizer, *pair))

    before_paired = [res(p) for p in GAME_PAIRS]
    protocol = machine.lexicon_game(GAME_PAIRS, rounds=15, alpha=0.3)
    after_paired = [res(p) for p in GAME_PAIRS]
    learned_rounds = sum(1 for r in protocol if r["learned"])
    mark("sp9_s6_regress",
         after_paired == SP6_AFTER_PAIRED and learned_rounds == 15)
    return {
        "before_paired": before_paired,
        "after_paired": after_paired,
        "learned_rounds": learned_rounds,
        "sp6_anchor": SP6_AFTER_PAIRED,
    }


def lane_saturation() -> Dict[str, Any]:
    """SP9.2 — ein Paar über 60 reine Episoden (α=0.3): Konvergenz, Monotonie,
    Kein-Overshoot; plus α-Sweep {0.1, 0.3, 0.9} über 100 reine Episoden und
    der Determinismus-Doppel-Check."""
    def run_once() -> List[float]:
        tokenizer = GeometricTokenizer()
        traj = []
        for _ in range(SAT_EPISODES):
            tokenizer.learn_pair("regel", "rule", alpha=0.3)
            traj.append(raw_res(tokenizer, "regel", "rule"))
        return traj

    traj_a = run_once()
    traj_b = run_once()
    dip = max_dip_after_peak(traj_a)
    first = first_at_threshold(traj_a)
    mark("sp9_det", traj_a == traj_b)
    mark("sp9_sat_converged", traj_a[-1] > THRESH)
    mark("sp9_sat_monotone", dip <= 1e-6 and max(traj_a) <= 1.0 + 1e-9)
    mark("sp9_sat_alpha3_first", first == 5)

    sweep: Dict[str, Any] = {}
    for alpha in ALPHAS:
        tokenizer = GeometricTokenizer()
        traj: List[float] = []
        for _ in range(SWEEP_EPISODES):
            tokenizer.learn_pair("regel", "rule", alpha=alpha)
            traj.append(raw_res(tokenizer, "regel", "rule"))
        sweep[repr(alpha)] = {
            "final": round6(traj[-1]),
            "conv_episode": first_at_threshold(traj),
            "traj": round6(traj),
        }
    slow = [first_at_threshold([float(v) for v in sweep[repr(alpha)]["traj"]])
            for alpha in (0.1, 0.9)]
    final_ok = all(sweep[repr(alpha)]["final"] > THRESH
                   for alpha in (0.1, 0.9))
    mark("sp9_sat_sweep_final", final_ok)
    traj01 = [float(v) for v in sweep[repr(0.1)]["traj"]]
    traj09 = [float(v) for v in sweep[repr(0.9)]["traj"]]
    sym = max(abs(a - b) for a, b in zip(traj01, traj09))
    mark("sp9_sat_sweep_symmetry", sym <= 1e-12)
    mid_fast = (sweep[repr(0.3)]["conv_episode"] is not None
                and sweep[repr(0.3)]["conv_episode"] <= 6
                and all(c is not None and c >= 12 for c in slow))
    mark("sp9_sat_sweep_mid_fast", mid_fast)
    return {
        "sat_alpha03": {
            "final": round6(traj_a[-1]),
            "dip_after_peak": round(dip, 12),
            "first_at_threshold": first,
            "overshoot": round(max(traj_a) - 1.0, 12),
            "traj": round6(traj_a),
        },
        "sweep": sweep,
        "sweep_symmetry_max": round(sym, 12),
        "sweep_conv_slow": slow,
    }


def lane_interference() -> Dict[str, Any]:
    """SP9.1 — Kette regel↔rule → rule↔game: geteiltes 'rule' zieht das
    gelernte Paar auseinander; das End-Embedding nähert sich der Halbierenden
    der Ursprungs-Vektoren. Disjunkt-Kontrast: kein geteiltes Wort → keine
    Bewegung."""
    tokenizer = GeometricTokenizer()
    regel0 = tokenizer.embed_word("regel")
    rule0 = tokenizer.embed_word("rule")
    game0 = tokenizer.embed_word("game")
    baseline_rule_game = raw_res(tokenizer, "regel", "game")

    # Phase 1: regel↔rule bis Konvergenz (ex ante: exakt 5 Episoden).
    used = 0
    peak = 0.0
    for i in range(100):
        tokenizer.learn_pair("regel", "rule", alpha=0.3)
        used = i + 1
        peak = raw_res(tokenizer, "regel", "rule")
        if peak >= THRESH:
            break

    # Phase-2-Anfangsstände — E1-Korrektur: auch der Beobachter 'regel' ist
    # selbst durch Phase 1 gedreht; die p4-Formel benutzt die TATSÄCHlichen
    # Phase-2-Stände statt der t0-Stände.
    regel_p2 = tokenizer.embed_word("regel")
    rule_p2 = tokenizer.embed_word("rule")

    # Phase 2: rule↔game, 30 reine Episoden; Beobachter: res(regel, rule).
    traj: List[float] = []
    for _ in range(CHAIN_EPISODES):
        tokenizer.learn_pair("rule", "game", alpha=0.3)
        traj.append(raw_res(tokenizer, "regel", "rule"))
    final_rule = raw_res(tokenizer, "regel", "rule")
    final_game = raw_res(tokenizer, "regel", "game")
    bisect_pred = bisector_cos(rule0, game0, regel0)      # Ursprungs-Lesung
    bisect_pred_p4 = bisector_cos(rule_p2, game0, regel_p2)  # Phase-2-Lesung
    mark("sp9_int_decay", final_rule <= peak - 0.05)
    mark("sp9_int_endpoint",
         abs(final_rule - bisect_pred) <= 0.05
         and abs(final_game - bisect_pred) <= 0.05)
    mark("sp9_int_endpoint_p4",
         abs(final_rule - bisect_pred_p4) <= 0.05
         and abs(final_game - bisect_pred_p4) <= 0.05)

    # Disjunkt-Kontrast: lernendes Tauschpaar ohne geteilte Wörter.
    disjoint = GeometricTokenizer()
    for _ in range(5):
        disjoint.learn_pair("regel", "rule", alpha=0.3)
    before_disjoint = raw_res(disjoint, "regel", "rule")
    for _ in range(CHAIN_EPISODES):
        disjoint.learn_pair("tisch", "stuhl", alpha=0.3)
    after_disjoint = raw_res(disjoint, "regel", "rule")
    mark("sp9_int_disjoint", abs(after_disjoint - before_disjoint) <= 1e-12)
    return {
        "baseline_rule_game": round6(baseline_rule_game),
        "phase1_episodes": used,
        "peak": round6(peak),
        "traj_phase2": round6(traj),
        "final_rule": round6(final_rule),
        "final_game": round6(final_game),
        "bisector_pred": round6(bisect_pred),
        "bisector_pred_p4": round6(bisect_pred_p4),
        "disjoint_delta": round(after_disjoint - before_disjoint, 12),
    }


def lane_unlearning() -> Dict[str, Any]:
    """SP9.3 — Verlernen durch Neugebrauch und Wiederherstellung durch
    wiederholten Gebrauch (kein Zerfallsterm im Canon)."""
    tokenizer = GeometricTokenizer()
    regel0 = tokenizer.embed_word("regel")
    rule0 = tokenizer.embed_word("rule")
    tisch0 = tokenizer.embed_word("tisch")

    for _ in range(5):
        tokenizer.learn_pair("regel", "rule", alpha=0.3)
    learned = raw_res(tokenizer, "regel", "rule")

    # Phase-2-Anfangsstände — dieselbe E1-Korrektur (der Beobachter 'regel'
    # friert in der Phase 2 ein, aber sein IST-Stand ist der Phase-2-Stand).
    regel_p2 = tokenizer.embed_word("regel")
    rule_p2 = tokenizer.embed_word("rule")

    traj_u: List[float] = []
    for _ in range(USE_EPISODES):
        tokenizer.learn_pair("rule", "tisch", alpha=0.3)
        traj_u.append(raw_res(tokenizer, "regel", "rule"))
    final_u = raw_res(tokenizer, "regel", "rule")
    bisect_u = bisector_cos(rule0, tisch0, regel0)       # Ursprungs-Lesung
    bisect_u_p4 = bisector_cos(rule_p2, tisch0, regel_p2)  # Phase-2-Lesung

    traj_r: List[float] = []
    relearn_at = None
    for i in range(RELEARN_LIMIT):
        tokenizer.learn_pair("regel", "rule", alpha=0.3)
        traj_r.append(raw_res(tokenizer, "regel", "rule"))
        if traj_r[-1] >= THRESH:
            relearn_at = i + 1
            break
    mark("sp9_unlearn_decay",
         final_u < 0.8 and abs(final_u - bisect_u) <= 0.02)
    mark("sp9_unlearn_endpoint_p4", abs(final_u - bisect_u_p4) <= 0.02)
    mark("sp9_unlearn_relearn", relearn_at is not None)
    return {
        "learned": round6(learned),
        "traj_use": round6(traj_u),
        "final_use": round6(final_u),
        "bisector_use_pred": round6(bisect_u),
        "bisector_use_pred_p4": round6(bisect_u_p4),
        "traj_relearn": round6(traj_r),
        "relearn_at": relearn_at,
    }


def lane_social() -> Dict[str, Any]:
    """SP9.4 — Sozial-Freeze: die unrelated-Resonanzen driften nur innerhalb
    des ersten Konvergenzfensters und frieren danach ein (RNG-Kontinuität
    über zwei lexicon_game-Legs derselben Maschine)."""
    machine = WittgensteinMachine(MachineConfig(seed=9, mode="in_loop"))
    tokenizer = machine.tokenizer

    def res(pair: Tuple[str, str]) -> float:
        return round6(raw_res(tokenizer, *pair))

    machine.lexicon_game(GAME_PAIRS, rounds=15, alpha=0.3)
    unrelated15 = [res(p) for p in UNRELATED_GAME]

    # Leg 2 runde-für-Runde: derselbe round-robin-Schedule und derselbe
    # RNG-Draw-Order wie eine einzelne lexicon_game(45)-Schrift (das Paar-
    # Loslassen verbraucht keine Zufallszüge), aber je Runde beobachtbar —
    # deshalb trägt die p4-Lesung die tatsächliche tisch-Trajektorie.
    tisch_traj = [float(unrelated15[0])]
    for r in range(45):
        w_lang, w_partner = GAME_PAIRS[r % len(GAME_PAIRS)]
        machine.lexicon_game([(w_lang, w_partner)], rounds=1, alpha=0.3)
        current = raw_res(tokenizer, *UNRELATED_GAME[0])
        tisch_traj.append(current)
    unrelated60 = [res(p) for p in UNRELATED_GAME]
    mean15 = float(np.mean(unrelated15))
    mean60 = float(np.mean(unrelated60))
    tisch15 = float(unrelated15[0])
    tisch60 = float(unrelated60[0])
    deltas = [b - a for a, b in zip(tisch_traj, tisch_traj[1:])]
    monotone = all(d >= 0.0 for d in deltas) or all(d <= 0.0 for d in deltas)
    drift_p4 = abs(tisch60 - tisch15)
    frozen = all(abs(unrelated60[i] - unrelated15[i]) == 0.0 for i in (1, 2))
    mark("sp9_social_freeze", abs(mean60 - mean15) <= 0.05)
    mark("sp9_social_tisch_pin",
         abs(tisch60 - tisch15) <= 1e-12
         and abs(round6(tisch15) - SP6_REGEL_TISCH_15) <= 1e-9)
    mark("sp9_social_tisch_p4", monotone and drift_p4 <= 2e-2)
    mark("sp9_social_frozen_exact", frozen)
    return {
        "unrelated15": unrelated15,
        "unrelated60": unrelated60,
        "mean15": round(mean15, 4),
        "mean60": round(mean60, 4),
        "tisch15": round6(tisch15),
        "tisch60": round6(tisch60),
        "tisch_traj_leg2": round6(tisch_traj),
        "tisch_deltas_nonzero": sum(1 for d in deltas if d != 0.0),
        "tisch_drift_p4": round(drift_p4, 9),
        "sp6_tisch_anchor": SP6_REGEL_TISCH_15,
    }

    report.update({"findings": findings, "all_pass": all_pass,

          f"/ {n_pass_fixed} korrigierte Lesungen")
    print("EXIT =", 0 if all_pass else 1)
    return 0 if all_pass else 1

FINDINGS: List[Tuple[str, str]] = []


def mark(name: str, ok: bool) -> bool:
    """Verdict zeichnen (Prereg-Namen; Reihenfolge = Findings-Reihenfolge)."""
    FINDINGS.append((name, "PASS" if ok else "FAIL"))
    return ok


def raw_res(tokenizer, w1: str, w2: str) -> float:
    return float(tokenizer.resonance(tokenizer.embed_word(w1),
                                     tokenizer.embed_word(w2)))


def dotv(u: List[float], v: List[float]) -> float:
    return float(sum(x * y for x, y in zip(u, v)))


def pair_geometry(tokenizer, w1: str, w2: str) -> Dict[str, Any]:
    """Gematria-Geometrie des ungelehrten Paars: c0, β0 = θ0/2, Mittelinie
    M = n(a+b), Achse N = n(a−b) (Seite von w1). Identität w1_0 = s/2 + d/2
    beweist die Zuschauer-Formel exakt."""
    a = tokenizer.embed_word(w1)
    b = tokenizer.embed_word(w2)
    c0 = float(tokenizer.resonance(a, b))
    sumv = [x + y for x, y in zip(a, b)]
    diff = [x - y for x, y in zip(a, b)]
    norm_s = math.sqrt(float(tokenizer.resonance(sumv, sumv)))
    norm_d = math.sqrt(float(tokenizer.resonance(diff, diff)))
    if norm_s < 1e-9 or norm_d < 1e-9:
        raise ValueError(f"pair_geometry: entartete Geometrie {w1}/{w2} "
                         f"(norms {norm_s:.3e}/{norm_d:.3e})")
    theta0 = math.acos(max(-1.0, min(1.0, c0)))
    return {"w1": w1, "w2": w2, "c0": c0, "beta0": theta0 / 2.0,
            "norm_sum": norm_s, "norm_diff": norm_d,
            "mid": [x / norm_s for x in sumv],
            "axis": [x / norm_d for x in diff]}


def closed_pair(geom: Dict[str, Any], episodes: int) -> float:
    """Paar-Resonanz nach `episodes` gelernten Episoden: c_j = cos(2β_j)."""
    beta = math.atan((SHRINK ** episodes) * math.tan(geom["beta0"]))
    return math.cos(2.0 * beta)


def closed_viewer(geom: Dict[str, Any], viewer: List[float],
                  episodes: int) -> float:
    """Zuschauer-Resonanz R_j = cos β_j (M·v) + sin β_j (N·v)."""
    beta = math.atan((SHRINK ** episodes) * math.tan(geom["beta0"]))
    return math.cos(beta) * dotv(geom["mid"], viewer) + \
        math.sin(beta) * dotv(geom["axis"], viewer)


def first_trace_diff(trace_a: List[str], trace_b: List[str]) -> Tuple[
        Optional[int], str]:
    """1-basiert der erste Differenzschritt; (None, …) wenn identisch."""
    for idx, (x, y) in enumerate(zip(trace_a, trace_b)):
        if x != y:
            return idx + 1, f"Schritt {idx + 1}: Spur-Differenz"
    if len(trace_a) != len(trace_b):
        return min(len(trace_a), len(trace_b)) + 1, "Laengen-Differenz"
    return None, "identisch"


def round6(values: List[float]) -> List[float]:
    return [round(v, 6) for v in values]


# ---------------------------------------------------------------------------
# Composite-Lauf-Harness (Zwangs-Haken am composite-step-Zähler)
# ---------------------------------------------------------------------------
def make_composite(force_ast: Tuple[Any, ...]) -> Tuple[CompositeMachine,
                                                        Dict[str, Any]]:
    """Composite (seed 7, SP7-Config) mit force-Haken: am Zwangsschritt wird
    die ERSTE Regel des Batch durch `force_ast` ersetzt — über einen
    pending-Flag, den die Lauf-Routine am composite-step-Zähler setzt, so
    dass Lexikon-Runden außerhalb der Composite-Schritte den Zwang nie fassen."""
    machine = CompositeMachine(MachineConfig(seed=SEED_KP, mode="in_loop",
                                             params=dict(TAUT_POOL6)))
    harness: Dict[str, Any] = {"pending": False}
    wm = machine.wittgenstein
    original = wm.create_rule

    def patched() -> GeneratedSymbolicRule:
        if harness["pending"]:
            harness["pending"] = False
            rule = GeneratedSymbolicRule(force_ast)
            wm.generated_rules.append(rule)
            return rule
        return original()

    wm.create_rule = patched
    return machine, harness


def run_composite_sp7(force_ast: Tuple[Any, ...],
                      force_steps: set) -> Dict[str, Any]:
    """Natur-Lage ohne Lexikon (L0/Kontroll-Form; SP7-Länge + volle Spur)."""
    machine, harness = make_composite(force_ast)
    verdicts: List[str] = []
    basis: List[int] = []
    gaps: List[float] = []
    paradigms: List[bool] = []
    for step in range(1, STEPS + 1):
        if step in force_steps:
            harness["pending"] = True
        result = machine.step(None)
        verdicts.append(str(result.invariants["verdict"]))
        basis.append(len(machine.turing.basis))
        gaps.append(float(result.gap))
        paradigms.append(bool(result.invariants.get(
            "paradigm_signal_composite", {}).get("detected")))
    first_detect = next((idx for idx, flag in enumerate(paradigms, start=1)
                         if flag), None)
    return {"verdicts": verdicts, "basis": basis, "final_basis": basis[-1],
            "expansions": list(machine.expansions), "gaps": gaps,
            "gaps_finite": all(g == g and abs(g) != float("inf") for g in gaps),
            "first_detect": first_detect,
            "trace": list(machine.trace)}


def run_forced(machine: CompositeMachine, harness: Dict[str, Any]
               ) -> Tuple[List[str], List[float]]:
    """SP7-Verlauf (Zwangs-Haken an FORCE_STEPS — identische Mechanik wie
    run_composite_sp7) für die Seitenband-Arme: gibt Spur + Gap-Rohwerte."""
    gaps: List[float] = []
    for step in range(1, STEPS + 1):
        if step in FORCE_STEPS:
            harness["pending"] = True
        result = machine.step(None)
        gaps.append(float(result.gap))
    return list(machine.trace), gaps

def lane_seitenband(anchor: Dict[str, Any]) -> Dict[str, Any]:
    """(a) Seitenband-Lexikon auf eigener Maschine (seed 9, SP6-Spiel) neben
    einem frischen Composite (seed 7, SP7-Zwang): Composite-Spur == L0-Spur,
    Lexikon == SP6-Anker (per Paar). (b) Lexikon auf DERSELBEN Composite-
    Maschine NACH dem 16-Schritt-Verlauf: Spur + Gap-Historie (Snapshot
    vor/nach) bitidentisch, Lexikon konvergiert."""
    side = WittgensteinMachine(MachineConfig(seed=SEED_SB, mode="in_loop"))
    protocol_side = side.lexicon_game(GAME_PAIRS, rounds=15, alpha=ALPHA)
    # SP6-Schreibweise: End-Resonanz je PAAR (nicht das Per-Runden-Protokoll).
    after_paired = [round(raw_res(side.tokenizer, *p), 6)
                    for p in GAME_PAIRS]
    protocol_res = [float(p["resonance_after"]) for p in protocol_side]
    learned_side = sum(1 for p in protocol_side if p["learned"])
    unrelated_side = [round(raw_res(side.tokenizer, *p), 6)
                      for p in UNRELATED_GAME]

    comp_a, harness_a = make_composite(TAUTAST)
    trace_a, gaps_a = run_forced(comp_a, harness_a)

    comp_b, harness_b = make_composite(TAUTAST)
    trace_b_before, gaps_b_before = run_forced(comp_b, harness_b)
    snapshot_before = json.dumps({"trace": trace_b_before,
                                  "gaps": list(comp_b._gap_history)})
    protocol_b = comp_b.wittgenstein.lexicon_game(GAME_PAIRS, rounds=15,
                                                  alpha=ALPHA)
    snapshot_after = json.dumps({"trace": list(comp_b.trace),
                                 "gaps": list(comp_b._gap_history)})
    final_b = [raw_res(comp_b.wittgenstein.tokenizer, *p) for p in GAME_PAIRS]

    # Erstrun-Konstruktionen (P3, s. ERRATA): Seitenband-Arm OHNE Zwang +
    # after_paired als Per-Runden-Protokoll — reproduzieren die FAIL-Lesung.
    comp_c, _ = make_composite(TAUTAST)
    for _ in range(STEPS):
        float(comp_c.step(None).gap)
    spur_erstrun = list(comp_c.trace) == anchor["trace"]
    lexikon_erstrun = protocol_res == SP6_AFTER_PAIRED

    anchor_trace, anchor_gaps = anchor["trace"], anchor["gaps"]
    spur_ok = (trace_a == anchor_trace and gaps_a == anchor_gaps
               and trace_b_before == trace_a and gaps_b_before == gaps_a
               and snapshot_before == snapshot_after)
    lexikon_ok = (after_paired == SP6_AFTER_PAIRED
                  and learned_side == 15
                  and all(r >= MIN_PAIR_RES for r in final_b))
    mark("sp10_t1_spur", spur_erstrun)       # Erstrun-Konstruktion (FAIL)
    mark("sp10_t1_lexikon", lexikon_erstrun)  # Erstrun-Konstruktion (FAIL)
    mark("sp10_t1_spur_p4", spur_ok)
    mark("sp10_t1_lexikon_p4", lexikon_ok)
    return {"after_paired": after_paired, "protocol_res": protocol_res,
            "learned_side": learned_side, "unrelated_side": unrelated_side,
            "trace_invariant": spur_ok, "trace_len": len(trace_a),
            "unforced_trace_matches": spur_erstrun,
            "final_b": round6(final_b),
            "learned_b": sum(1 for p in protocol_b if p["learned"])}

# L2: gezählter Zufalls-Vorschub (These 2)
# ---------------------------------------------------------------------------
def lane_counted(force_ast: Tuple[Any, ...],
                 anchor_trace: List[str]) -> Dict[str, Any]:
    """Verschachtelung: je Composite-Schritt 3 Lexikon-Runden (round-robin
    über GAME_PAIRS) auf DEMSELBEN Wittgenstein-RNG-Strom — Composite-Schritt
    zuerst, Lexikon-Runden danach (der erste Composite-Schritt zieht noch aus
    der unverschobenen Position, die Divergenz erscheint also naturwüchsig
    ab dem Schritt nach dem ersten Vorschub)."""
    machine, harness = make_composite(force_ast)
    wm = machine.wittgenstein
    tok = wm.tokenizer
    state_fresh = json.dumps(wm._rng.bit_generator.state, sort_keys=True)
    geoms = {pair: pair_geometry(tok, *pair) for pair in GAME_PAIRS}
    viewer = tok.embed_word("tisch")
    pre_rows = [raw_res(tok, *pair) for pair in UNRELATED_GAME]
    episodes = [0, 0, 0]
    lex_rows: List[Dict[str, Any]] = []
    rows_series: List[List[float]] = []   # je Spielrunde die 3 Zeilen (roh)
    verdicts: List[str] = []
    basis: List[int] = []
    gaps: List[float] = []
    lex_i = 0
    for step in range(1, STEPS + 1):
        if step in FORCE_STEPS:
            harness["pending"] = True
        result = machine.step(None)
        verdicts.append(str(result.invariants["verdict"]))
        basis.append(len(machine.turing.basis))
        gaps.append(float(result.gap))
        for _ in range(LEX_RPS):
            pair = GAME_PAIRS[lex_i % len(GAME_PAIRS)]
            line = wm.lexicon_game([pair], rounds=1, alpha=ALPHA)[0]
            if line["learned"]:
                episodes[lex_i % len(GAME_PAIRS)] += 1
            k = episodes[lex_i % len(GAME_PAIRS)]
            rows_now = [raw_res(tok, *p) for p in UNRELATED_GAME]
            lex_rows.append({
                "round": lex_i, "pair": list(pair),
                "learned": bool(line["learned"]), "episodes": k,
                "res": raw_res(tok, *pair),
                "closed": closed_pair(geoms[pair], k),
                "success": line["success_rate"], "k0": episodes[0]})
            rows_series.append(rows_now)
            lex_i += 1
    rows_post = [raw_res(tok, *p) for p in UNRELATED_GAME]
    pair_final = [raw_res(tok, *p) for p in GAME_PAIRS]
    trace = list(machine.trace)
    first_diff, diff_note = first_trace_diff(trace, anchor_trace)
    return {"force_ast": str(force_ast), "verdicts": verdicts,
            "expansions": list(machine.expansions),
            "final_basis": basis[-1], "gaps": gaps,
            "gap_finite": all(g == g and abs(g) != float("inf") for g in gaps),
            "trace": trace,
            "first_diff": first_diff, "diff_note": diff_note,
            "rng_shifted_after_lexikon": json.dumps(
                wm._rng.bit_generator.state, sort_keys=True) != state_fresh,
            "episodes": list(episodes),
            "pair_final": round6(pair_final),
            "rows_pre": round6(pre_rows), "rows_post": round6(rows_post),
            "_rows_pre": list(pre_rows), "_rows_post": list(rows_post),
            "lex_rounds": len(lex_rows),
            "_lex_rows": lex_rows, "_rows_series": rows_series,
            "_geoms": geoms, "_viewer": viewer}


def lane_counted_pair(anchor_trace: List[str]) -> Tuple[Dict[str, Any],
                                                        Dict[str, Any]]:
    """L2: kripke-Arm (Doppel-Lauf) + quus-Arm (Doppel-Lauf); Marks:
    t2_divergenz, t2_kripke, t2_lexikon_lebt; Determinismus-Fracht det2."""
    k1 = lane_counted(TAUTAST, anchor_trace)
    k2 = lane_counted(TAUTAST, anchor_trace)
    q1 = lane_counted(QUUSAST, anchor_trace)
    q2 = lane_counted(QUUSAST, anchor_trace)
    det2 = (k1["trace"] == k2["trace"] and k1["gaps"] == k2["gaps"]
            and q1["trace"] == q2["trace"] and q1["gaps"] == q2["gaps"]
            and [r["res"] for r in k1["_lex_rows"]]
            == [r["res"] for r in k2["_lex_rows"]]
            and k1["_rows_series"] == k2["_rows_series"])
    step9 = STEP9
    kripke_ok = (k1["verdicts"][step9 - 1] == "LOOP"
                 and "step9:LOOP" in k1["expansions"]
                 and k1["final_basis"] > 3 and k1["gap_finite"])
    quus_ok = (q1["verdicts"][step9 - 1] != "LOOP"
               and "step9:LOOP" not in q1["expansions"])
    divergenz_ok = k1["first_diff"] is not None and k1["first_diff"] <= 3
    lexikon_lebt = (all(r >= MIN_PAIR_RES for r in k1["pair_final"])
                    and all(n >= MIN_EPISODES for n in k1["episodes"]))
    mark("sp10_t2_divergenz", divergenz_ok)
    mark("sp10_t2_kripke", kripke_ok and quus_ok)
    mark("sp10_t2_lexikon_lebt", lexikon_lebt)
    summary = {
        "kripke": {key: k1[key] for key in
                   ("verdicts", "expansions", "final_basis", "first_diff",
                    "diff_note", "rng_shifted_after_lexikon", "episodes",
                    "pair_final", "rows_pre", "rows_post", "lex_rounds",
                    "gap_finite")},
        "quus": {key: q1[key] for key in
                 ("verdicts", "expansions", "final_basis", "episodes",
                  "pair_final")},
        "det2": det2, "kripke_ok": kripke_ok, "quus_ok": quus_ok,
    }
    return summary, {"_k1": k1, "_q1": q1}


# ---------------------------------------------------------------------------
# L3b: Analytik der L2-Lage (These 3, LIVE)
# ---------------------------------------------------------------------------
def lane_analytik(k1: Dict[str, Any]) -> Dict[str, Any]:
    """Geschlossene Formen gegen die L2-Rohwerte: Paar-Resonanz über alle
    48 Spielrunden (c_j je Runde gegen den Rohwert NACH der Runde), tisch-
    Zuschauer-Spur (49 Punkte: Punkt 0 = Pre-Lage mit k = 0 vor dem ersten
    Vorschub, Punkte 1..48 = die Zeilenlage je Runde mit der bis dahin
    gelernten regel↔rule-Episodenzahl), Zeilen-Konstanz für kaffee/baum,
    Grenzwert R_∞ = M·v."""
    geoms, viewer = k1["_geoms"], k1["_viewer"]
    lex_rows = k1["_lex_rows"]
    rows_series = k1["_rows_series"]
    rows_pre = k1["_rows_pre"]
    rows_post = k1["_rows_post"]

    # 1) Paar-Formel je Spielrunde (der Rohwert ist immer NACH der Runde
    #    genommen, `episodes` zählt ein ggf. Lernen derselben Runde).
    pair_devs = [abs(r["res"] - r["closed"]) for r in lex_rows]
    max_pair = max(pair_devs)

    # 2) tisch-Spur: Punkt 0 (k = 0) + 48 Runden; k = gelernte regel↔rule-
    #    Episoden (= lex_rows[i]["k0"], Buchhaltung des ersten Paares).
    regel_geom = geoms[GAME_PAIRS[0]]
    k0_series = [0] + [row["k0"] for row in lex_rows]
    tisch_series = [rows_pre[0]] + [rs[0] for rs in rows_series]
    obs_devs = [abs(R - closed_viewer(regel_geom, viewer, k))
                for R, k in zip(tisch_series, k0_series)]
    max_obs = max(obs_devs)

    # 3) kaffee/baum-Zeilen bitkonstant über alle Runden und in der Endlage.
    frozen_ok = (all(rs[1] == rows_pre[1] and rs[2] == rows_pre[2]
                     for rs in rows_series)
                 and rows_post[1] == rows_pre[1]
                 and rows_post[2] == rows_pre[2])

    # 4) Grenzwert: die letzte tisch-Lage liegt im 1e-4-Band um M·v.
    asym = abs(tisch_series[-1] - dotv(regel_geom["mid"], viewer))

    t3_paar = mark("sp10_t3_paar", max_pair <= ANALYTIC_TOL)
    t3_obs = mark("sp10_t3_obs", max_obs <= ANALYTIC_TOL and frozen_ok)
    return {"max_pair_dev": max_pair, "max_obs_dev": max_obs,
            "frozen_ok": frozen_ok, "t3_paar": t3_paar, "t3_obs": t3_obs,
            "asym": asym, "asym_reference": dotv(regel_geom["mid"], viewer),
            "tisch_series": round6(tisch_series), "k0_series": k0_series}

# L3a: SP9-Rückdiction
# ---------------------------------------------------------------------------
def lane_rueckdiction() -> Dict[str, Any]:
    committed = json.loads((REPO_ROOT / "scratches" /
                            "lexikon_dynamik_experiment.json").read_text(
                                encoding="utf-8"))
    soc = committed["social"]
    machine = WittgensteinMachine(MachineConfig(seed=SEED_SB, mode="in_loop"))
    tok = machine.tokenizer
    geoms = {pair: pair_geometry(tok, *pair) for pair in GAME_PAIRS}
    viewer = tok.embed_word("tisch")
    episodes = [0, 0, 0]
    pair_rows: List[Dict[str, Any]] = []

    # Leg 1: 15 gated Runden (einzelner lexicon_game-Aufruf, wie SP9).
    protocol = machine.lexicon_game(GAME_PAIRS, rounds=15, alpha=ALPHA)
    for line in protocol:
        pair = tuple(line["pair"])
        pi = GAME_PAIRS.index(pair)
        if line["learned"]:
            episodes[pi] += 1
        pair_rows.append({"round": int(line["round"]), "leg": 1,
                          "pair": list(pair), "learned": bool(line["learned"]),
                          "episodes": episodes[pi],
                          "res": None, "closed": None})
    unrelated15 = [round(raw_res(tok, *p), 6) for p in UNRELATED_GAME]
    # SP9-Schreibweise: der Spur-Anchluss ist der gerundete SP6-Anker
    # (unrelated15[0]); die Rohspur kommt daneben für die geschlossene Form.
    traj: List[float] = [unrelated15[0]]
    traj_raw: List[float] = [raw_res(tok, *UNRELATED_GAME[0])]
    k_list: List[int] = [episodes[0]]

    # Leg 2: 45 Einzel-Runden (wie SP9, je Runde beobachtbar).
    for r in range(45):
        pair = GAME_PAIRS[r % len(GAME_PAIRS)]
        line = machine.lexicon_game([pair], rounds=1, alpha=ALPHA)[0]
        if line["learned"]:
            episodes[r % len(GAME_PAIRS)] += 1
        k = episodes[r % len(GAME_PAIRS)]
        pair_rows.append({"round": r, "leg": 2, "pair": list(pair),
                          "learned": bool(line["learned"]),
                          "episodes": k,
                          "res": raw_res(tok, *pair),
                          "closed": closed_pair(geoms[pair], k)})
        row = raw_res(tok, *UNRELATED_GAME[0])
        traj.append(row)
        traj_raw.append(row)
        k_list.append(episodes[0])

def signal_from_history(history: Sequence[float]
                        ) -> Tuple[Optional[float], bool]:
    """Exakte Replikation: am Beobachtungsschritt sieht das COMPOSITE-Signal
    die Serie der Schritte 1..n-1 (Append nach der Prüfung, :143); das
    GOEDEL-Signal die Serie 1..n (Goedel.step läuft vor :135)."""
    if len(history) < 10:
        return None, False
    recent = float(np.mean(history[-5:]))
    baseline = (float(np.mean(history[-20:-5])) if len(history) >= 20
                else float(np.mean(history[:-5])))
    ratio = recent / (baseline + GAP_EPS)
    return ratio, bool(ratio > THRESHOLD)


def replicate_flags(series: Sequence[float], offset: int) -> List[Tuple[Optional[float], bool]]:
    """(ratio, detected) je Beobachtungsschritt n=1..len(series);
    offset der Anzahl Gaps, die das Signal im Schritt N schon sieht:
    Composite-Signal: offset 0 (sieht Serie bis n-1), Goedel-Signal: offset 1."""
    out: List[Tuple[Optional[float], bool]] = []
    for n in range(1, len(series) + 1):
        history = series[:n - 1 + offset]
        out.append(signal_from_history(history))
    return out

def make_composite(seed: int, force_ast: Tuple[Any, ...]
                   ) -> Tuple[CompositeMachine, Dict[str, Any]]:
    """Composite mit Zwangs-Haken am composite-step-Zähler (pending-Flag):
    die ERSTE Regel des Batch wird am Zwangsschritt durch force_ast ersetzt —
    ohne eine Zeile Maschinen-Code (wm.create_rule-Abdeckung, SP10-Muster)."""

                                             params=dict(POOL6)))
    harness: Dict[str, Any] = {"pending": False}
    wm = machine.wittgenstein

            harness["pending"] = False

        return original()

    return machine, harness


def run_composite(machine: CompositeMachine, harness: Dict[str, Any],
                  steps: int, force_steps: "set[int]") -> Dict[str, Any]:
    """Verlauf mit je-Schritt-Attribution: Sub-Gaps, beide Signale (goedel /
    composite), neue expansions-Einträge, Basis-Größe."""
    rows: List[Dict[str, Any]] = []
    trace: List[str] = []
    for step in range(1, steps + 1):
        if step in force_steps:
            harness["pending"] = True
        pre_expansions = list(machine.expansions)
        st = machine.step(None)
        inv = st.invariants
        new_exp = [e for e in machine.expansions if e not in pre_expansions]
        trace.append(st.trace[-1])
        rows.append({
            "step": step,
            "rule_name": inv["rule_name"],
            "verdict": inv["verdict"],
            "transitions": int(inv["transitions"]),
            "gap": float(st.gap),
            "w_gap": float(inv["submachines"]["wittgenstein"]["gap"]),
            "t_gap": float(inv["submachines"]["turing"]["gap"]),
            "g_gap": float(inv["submachines"]["goedel"]["gap"]),
            "g_ratio": (float(inv["paradigm_signal"]["ratio"])
                        if inv["paradigm_signal"]["ratio"] is not None else None),
            "g_detected": bool(inv["paradigm_signal"]["detected"]),
            "c_ratio": (float(inv["paradigm_signal_composite"]["ratio"])
                        if inv["paradigm_signal_composite"]["ratio"] is not None else None),
            "c_detected": bool(inv["paradigm_signal_composite"]["detected"]),
            "new_expansions": new_exp,
            "basis_size": len(machine.turing.basis),
        })
    expansions = list(machine.expansions)
    verdict_exp_steps = sorted(
        int(m.group(1)) for m in
        (re.match(r"step(\d+):(LOOP|COLLAPSE|EXPLOSION)", e) for e in expansions)
        if m)
    para_steps = sorted(
        int(m.group(1)) for m in
        (re.match(r"step(\d+):paradigm", e) for e in expansions) if m)
    gap_series = [r["gap"] for r in rows]
    w_series = [r["w_gap"] for r in rows]
    t_series = [r["t_gap"] for r in rows]
    g_series = [r["g_gap"] for r in rows]
    return {
        "rows": rows, "trace": trace, "expansions": expansions,
        "verdict_exp_steps": verdict_exp_steps, "para_steps": para_steps,
        "gap_series": gap_series, "w_series": w_series, "t_series": t_series,
        "g_series": g_series, "basis_size": rows[-1]["basis_size"],
        "final_basis": len(machine.turing.basis),
    }

    """T2-Replikation: Composite-Detect-Muster aus der finalen Gap-Serie

def counterfactuals(run: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    """T3b/T3c — Gefrieren auf dem Vor-Expansions-Niveau, Datenebene.
    None, wenn der Lauf keine Expansion hatte."""
    first = run["verdict_exp_steps"][0] if run["verdict_exp_steps"] else None
    if not first:
        return None
    w, t, g = run["w_series"], run["t_series"], run["g_series"]
    cut = first - 1
    t_pre = float(np.mean(t[:cut]))
    g_pre = float(np.mean(g[:cut]))
    cf_t = [t[i] if i < cut else t_pre for i in range(len(t))]
    cf_g = [g[i] if i < cut else g_pre for i in range(len(g))]
    cf_b = [(w[i] + cf_t[i] + g[i]) / 3.0 for i in range(len(w))]
    cf_c = [(w[i] + t[i] + cf_g[i]) / 3.0 for i in range(len(w))]
    return {
        "first_expansion": first,
        "t_pre": t_pre, "g_pre": g_pre,
        "orig_detections": list(run["para_steps"]),
        "cf_b_detections": [n for n, (_, det) in
                            enumerate(replicate_flags(cf_b, offset=0), 1) if det],
        "cf_c_detections": [n for n, (_, det) in
                            enumerate(replicate_flags(cf_c, offset=0), 1) if det],
        "orig_flags": [r["c_detected"] for r in run["rows"]],
        "cf_b_flags": [det for _, det in replicate_flags(cf_b, offset=0)],
        "cf_c_flags": [det for _, det in replicate_flags(cf_c, offset=0)],
    }


def t2_check(run: Dict[str, Any]) -> Tuple[bool, str]:
    """Mechanik-Falsifikator: rekomputierte (ratio, detected) == aufgezeichnet."""
    rep = rep_composite_flags(run)
    for row, (ratio, det) in zip(run["rows"], rep):

            return False, f"Flag-Differenz am Schritt {row['step']}"
        if (ratio is None) != (row["c_ratio"] is None):
            return False, f"Ratio-None-Differenz am Schritt {row['step']}"
        if ratio is not None and abs(ratio - row["c_ratio"]) > RATIO_TOL:
            return False, (f"Ratio-Differenz am Schritt {row['step']}: "
                           f"{ratio!r} != {row['c_ratio']!r}")
    # Detect-Menge-Ableitung aus den expansions-Einträgen (Konsistenz):
    return True, "ok"
