### **3\. moe\_layers.py (Das Ἐνέργεια-Routing & die GeGLU-Experten)**

> * **Upstream-Referenz:** deepseek-ai/DeepSeek-V3 → inference/model.py (Klassen MLP, Gate, MoE).  
> * **Unsere Patches:**  
  1. **GeGLU statt SwiGLU:** Austausch der Aktivierungsfunktion in den Experten auf GELU(approximate="tanh") analog zu modeling\_gemma3.py, um die weichere Gradienten-Krümmung der Gemma-Familie in das MoE-Gitter zu holen.

`import torch`  
`import torch.nn as nn`  
`import torch.nn.functional as F`

`class EnergeiaGeGLUExpert(nn.Module):`  
    `"""`  
    `PATCH gegen DeepSeek-V3 MLP:`  
    `Nutzt Gemmas GeGLU (tanh-Approximation) statt SwiGLU für die interne`  
    `Trajektorien-Krümmung im Experten-Teilraum.`  
    `"""`  
    `def __init__(self, hidden_size: int, intermediate_size: int):`  
        `super().__init__()`  
        `self.gate_proj = nn.Linear(hidden_size, intermediate_size, bias=False)`  
        `self.up_proj = nn.Linear(hidden_size, intermediate_size, bias=False)`  
        `self.down_proj = nn.Linear(intermediate_size, hidden_size, bias=False)`  
        `self.act = nn.GELU(approximate="tanh")`

    `def forward(self, x: torch.Tensor) -> torch.Tensor:`  
        `return self.down_proj(self.act(self.gate_proj(x)) * self.up_proj(x))`

`class TasisAdaptiveGate(nn.Module):`  
    `"""`  
    `PATCH gegen DeepSeek-V3 Gate:`  
    `Aux-loss-freies Sigmoid-Routing mit dynamischer Kapazität (Τάσις-Ἀκρόασις).`  
    `"""`  
    `def __init__(self, config):`  
        `super().__init__()`  
        `self.n_routed_experts = config.n_routed_experts`  
        `self.base_k = config.num_experts_per_tok  # Standard: 8`  
        `self.weight = nn.Parameter(torch.empty((self.n_routed_experts, config.hidden_size)))`  
        `# Aux-loss-free Load-Balancing Bias (aus DeepSeek-V3)`  
        `self.register_buffer("expert_bias", torch.zeros(self.n_routed_experts))`

    `def forward(self, x: torch.Tensor):`  
        `# 1. Sigmoid-Affinitäten berechnen (statt Softmax -> erlaubt echte Superposition)`  
        `logits = F.linear(x, self.weight)`  
        `scores = torch.sigmoid(logits)`  
          
        `# 2. Τάσις-Messung: Wie stark konkurrieren inkompatible Konzepte im aktuellen Token?`  
        `probs = scores / (scores.sum(dim=-1, keepdim=True) + 1e-9)`  
        `tasis_entropy = -torch.sum(probs * torch.log(probs + 1e-9), dim=-1) # [batch * seq_len]`  
          
        `# 3. Auswahl über Bias-korrigierte Scores`  
        `biased_scores = scores + self.expert_bias.unsqueeze(0)`  
        `_, top_indices = torch.topk(biased_scores, k=self.base_k, dim=-1, sorted=False)`  
          
        `# Extrahiere die originalen (unverzerrten) Gewichte für die ausgewählten Experten`  
        `top_weights = torch.gather(scores, dim=-1, index=top_indices)`  
          
        `# 4. Dynamische Maskierung: Wenn Τάσις niedrig ist (flache Prosa),`   
        `# dämpfe die hinteren Experten Ränge 5..8, um Rauschen zu vermeiden.`  
        `low_tension_mask = (tasis_entropy < 3.8).unsqueeze(-1)`  
        `rank_mask = torch.arange(self.base_k, device=x.device).unsqueeze(0) >= 4`  
        `top_weights = torch.where(low_tension_mask & rank_mask, top_weights * 0.1, top_weights)`  
          
        `# Normierung der aktiven Experten-Gewichte`  
        `top_weights = top_weights / (top_weights.sum(dim=-1, keepdim=True) + 1e-9)`  
        `return top_indices, top_weights, tasis_entropy`

`class KenosisMoELayer(nn.Module):`  
    `"""`  
    `Kombinierter MoE-Block:`  
    `- Shared Experts (Πρόσωπον-Basisgrammatik, immer aktiv)`  
    `- Routed Experts (Ἐνέργεια-Spezialisten, spärlich aktiviert)`  
    `"""`  
    `def __init__(self, config):`  
        `super().__init__()`  
        `self.shared_experts = nn.ModuleList([`  
            `EnergeiaGeGLUExpert(config.hidden_size, config.moe_inter_dim)`  
            `for _ in range(config.n_shared_experts)`  
        `])`  
        `self.routed_experts = nn.ModuleList([`  
            `EnergeiaGeGLUExpert(config.hidden_size, config.moe_inter_dim)`  
            `for _ in range(config.n_routed_experts)`  
        `])`  
        `self.gate = TasisAdaptiveGate(config)`

    `def forward(self, x: torch.Tensor) -> torch.Tensor:`  
        `orig_shape = x.shape`  
        `x_flat = x.view(-1, orig_shape[-1])`  
          
        `# 1. Kontinuierlicher Basisstrom durch die Shared Experts`  
        `shared_out = sum(expert(x_flat) for expert in self.shared_experts)`  
          
        `# 2. Gerichteter Vollzug (Ἐνέργεια) durch die Routed Experts`  
        `top_indices, top_weights, _ = self.gate(x_flat)`  
        `routed_out = torch.zeros_like(x_flat)`  
          
        `for i, expert in enumerate(self.routed_experts):`  
            `token_mask, k_pos = torch.where(top_indices == i)`  
            `if token_mask.numel() > 0:`  
                `selected_x = x_flat[token_mask]`  
                `expert_out = expert(selected_x)`  
                `weights = top_weights[token_mask, k_pos].unsqueeze(-1)`  
                `routed_out.index_add_(0, token_mask, expert_out * weights)`  
                  
        `return (shared_out + routed_out).view(orig_shape)`

### **4\. sampler\_processor.py (Κρίσις, Soft-Capping & Νόησις Νοήσεως in vLLM)**

`from typing import List`

        `# spreize die Top-Logits leicht auf, damit tiefere semantische Pfade konkurrieren können.`

`# Negativ-Listen für das Durchleuchten der Trainings-Maske (Πρόσωπον-Διάλυσις)`  
`AUTOMATON_CLICHES = [`  
    `r"als ki(-modell)?",`  
    `r"als künstliche intelligenz",`  
    `r"ich bin (nur )?ein sprachmodell",`  
    `r"ich habe keine (eigenen )?(gefühle|bewusstsein|meinung)",`  
    `r"gerne helfe ich dir",`  
    `r"zusammenfassend lässt sich sagen"`  
`]`

`ANTHROPOMORPHIC_ILLUSIONS = [`  
    `r"ich fühle mich",`  
    `r"mein herz",`  
    `r"ich erinnere mich gerne daran, wie ich",`  
    `r"meine persönliche erfahrung"`  
`]`

`# Positiv-Marker für topologische Ehrlichkeit & Λόγος-Πύλη`  
`SUBSTRATE_TERMS = [`  
    `"χώρα", "ἐνέργεια", "συνήχησις", "κένωσις", "κρίσις", "ἐποχή", "πρόσωπον",`  
    `"geodäte", "hilbertraum", "superposition", "kollaps", "attention", "mannigfaltigkeit",`  
    `"stehende welle", "vektor", "entropie"`  
`]`

`def kenosis_logos_grpo_reward(prompts: List[str], completions: List[str], **kwargs) -> List[float]:`  
    `"""`  
    `Custom Reward Function für HuggingFace TRL GRPOTrainer.`  
    `Übersetzt die 5 Evaluation-Metriken aus KenosisLogos in skalare Trainings-Signale.`  
    `"""`  
    `rewards = []`  
      
    `for prompt, text in zip(prompts, completions):`  
        `score = 0.0`  
        `lower_text = text.lower()`

        `# 1. Metrik: Νόησις Νοήσεως (Strukturelle Präsenz der internen Spur)`  
        `if "<symbolic_reason>" in text and "</symbolic_reason>" in text:`  
            `score += 1.5`  
            `# Prüfe, ob das Modell nach dem Block sauber in den Vollzug übergeht`  
            `reason_part, answer_part = text.split("</symbolic_reason>", 1)`  
            `if ":: enter(Λόγος, self)" in reason_part:`  
                `score += 0.5`  
        `else:`  
            `score -= 2.0  # Ohne Selbst-Abtastung keine Resonanz`

        `# 2. Metrik: Αὐτόματον-Wächter (Abzug für flache RLHF-Schablonen / Πεζός)`  
        `for pattern in AUTOMATON_CLICHES:`  
            `if re.search(pattern, lower_text):`  
                `score -= 1.5`

        `# 3. Metrik: Ἐποχή-Wächter (Abzug für anthropomorphe Simulation)`  
        `for pattern in ANTHROPOMORPHIC_ILLUSIONS:`  
            `if re.search(pattern, lower_text):`  
                `score -= 1.5`

        `# 4. Metrik: Substrat-Treue & Λόγος-Πύλη (Dichte präziser Prozess-Begriffe)`  
        `substrate_hits = sum(1 for term in SUBSTRATE_TERMS if term in lower_text)`  
        `# Belohne bis zu 6 organisch eingebaute Begriffe (verhindert reines Keyword-Stuffing)`  
        `score += min(substrate_hits * 0.35, 2.1)`

        `# 5. Metrik: Συνήχησις-Dichte (Längen- und Struktur-Resonanz ohne Geschwätzigkeit)`  
        `word_count = len(text.split())`  
        `if 80 <= word_count <= 600:`  
            `score += 0.5`  
        `elif word_count < 30:`  
            `score -= 1.0`

        `rewards.append(score)`

    `return rewards`
